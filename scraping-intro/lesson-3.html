import requests
from bs4 import BeautifulSoup
from urllib.parse import urljoin

listing_url = BASE_URL + 'practice/frogs-1.html'
response = requests.get(listing_url, timeout=10)
response.raise_for_status()
soup = BeautifulSoup(response.text, 'html.parser')
links = soup.select('a.frog-link')

records = []

for link in links:
    detail_url = urljoin(listing_url, link.get('href'))
    detail_response = requests.get(detail_url, timeout=10)
    detail_response.raise_for_status()
    detail_soup = BeautifulSoup(detail_response.text, 'html.parser')

    family_element = detail_soup.select_one('.frog-family')
    family = family_element.get_text(strip=True)

    records.append({
        'frog': link.get_text(strip=True),
        'family': family,
        'source_url': detail_url,
    })

print(len(records))
print(records)

Hint: excluding Previous and Next

Their classes are page-prev and page-next. The numbered links only have page-link. A selector such as .pagination a.page-link:not(.page-prev):not(.page-next) keeps the three numbered ones. Filtering on the visible text being 1, 2 or 3 works too.

import requests
from bs4 import BeautifulSoup
from urllib.parse import urljoin

listing_url = BASE_URL + 'practice/frogs-1.html'
response = requests.get(listing_url, timeout=10)
response.raise_for_status()
soup = BeautifulSoup(response.text, 'html.parser')

page_links = soup.select('.pagination a.page-link:not(.page-prev):not(.page-next)')
listing_urls = [urljoin(listing_url, link.get('href')) for link in page_links]

print(listing_urls)

records = []
for url in listing_urls:
    listing_response = requests.get(url, timeout=10)
    listing_response.raise_for_status()
    listing_soup = BeautifulSoup(listing_response.text, 'html.parser')

    for link in listing_soup.select('a.frog-link'):
        records.append({
            'frog': link.get_text(strip=True),
            'source_url': urljoin(url, link.get('href')),
        })

print(len(records))
print(len({row['source_url'] for row in records}))
